    for name in (f'{NAME}.md', f'{NAME}.html', f'{NAME}-agent.md', 'path-preprocessing-prefit-20260908.md'):

               'build_path_generalization_report.py')
    for name in scripts:
        files[f'scripts/{name}'] = (ROOT / 'scripts' / name).read_bytes()
    for directory in ('gated-export-round1', 'final-inference'):
        for path in sorted((ROOT / 'work/path-autoresearch-20260908/code-snapshots' / directory).iterdir()):
            if path.is_file():
                files[f'historical-snapshots/{directory}/{path.name}'] = path.read_bytes()
    for name in ('verification-final.json', 'descriptive-audit.json', 'report-validation.json'):
        files[f'receipts/{name}'] = (RUN / name).read_bytes()
    files['receipts/final_summary.json'] = (RUN / 'final-20260908T133200Z/final_summary.json').read_bytes()
    for name in ('PROTOCOL-FROZEN-V1.md', 'prefit-selected-sources.json', 'ROUND2-DECISION.md', 'SELECTION-FROZEN.json',
                 'science-prefit-v1-review.md', 'code-review-round1-gate.md', 'round1-evaluator-round2-gate.md',
                 'final-code-gate-followup.md', 'final-science-results-review.md', 'final-teaching-release-review.md',
                 'final-code-evidence-review.md', 'final-release-followup.md', 'REVIEW-RESOLUTION.md',
                 'FINAL-ATTEMPT-1-FAILURE.json'):
        files[f'receipts/{name}'] = (MANAGER / name).read_bytes()
    for name in ('summarize_final_evidence.py', 'check_report_delivery.py', 'make_portable_bundle.py'):
        files[f'delivery-checks/{name}'] = (ROOT / 'work/path-autoresearch-20260908' / name).read_bytes()
    readme = f'''# Portable path-preprocessing report/code package — 2026-09-08

**Not a full analysis backup:** raw cue data, source tensors, all saved models/predictions/lineages and runtime environments are NOT included. They remain on the data-equipped host under:

```sh

  --run-dir outputs/path-preprocessing-autoresearch-20260908/biological-20260908T115508Z \\
```

    files['README.md'] = readme.encode()
    manifest = {name: hashlib.sha256(data).hexdigest() for name, data in files.items()}
    files['SHA256-MANIFEST.json'] = (json.dumps(manifest, indent=2, sort_keys=True) + '\n').encode()

        for name, data in sorted(files.items()):
            assert not Path(name).is_absolute() and '..' not in Path(name).parts

    with zipfile.ZipFile(destination) as archive:
        assert archive.testzip() is None
        assert len(archive.namelist()) == len(files)
        for name, expected in manifest.items():
            assert hashlib.sha256(archive.read(name)).hexdigest() == expected
    receipt = {'status': 'byte-verified', 'new_model_fits': 0, 'zip': str(destination),
