css = '''body{font:16px/1.6 system-ui,sans-serif;color:#17324a;background:#f6f8fb;margin:0}main{max-width:1250px;margin:auto;padding:28px}a{color:#075d9b}code{font-family:ui-monospace,monospace;overflow-wrap:anywhere}pre{padding:12px;background:white;border:1px solid #ccd6e0;font:13px/1.5 ui-monospace,monospace;white-space:pre-wrap}.line{display:flex;min-height:1.5em}.line code{flex:1;min-width:0;white-space:pre-wrap}.n{width:4.5em;flex-shrink:0;color:#738294;text-decoration:none}.line:target{background:#fff0b8}li{margin:.55em 0}.note{border-left:4px solid #a4522a;padding:12px 18px;background:#fff0e8}small{overflow-wrap:anywhere}h1,h2{line-height:1.2}'''


def page(title, body):
    return '<!doctype html><html lang="en"><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><meta http-equiv="Content-Security-Policy" content="default-src \'none\'; style-src \'unsafe-inline\'; img-src data:; base-uri \'none\'"><title>'+escape(title)+'</title><style>'+css+'</style><main>'+body+'</main></html>'


assert not out.exists(), 'Refusing to overwrite an existing audit-view directory'
out.mkdir(parents=True)
manifest = []
sections = {}
for group, alias, title, rel in entries:
    source = Path(rel) if Path(rel).is_absolute() else repo / rel
    assert source.is_file(), source
    data = source.read_bytes()
    sha = hashlib.sha256(data).hexdigest()
    destination = out / (alias + '.html')
    if source.suffix == '.html':
        destination.write_bytes(data)
    else:
        lines = data.decode('utf-8').splitlines()
        content = ''.join('<span class="line" id="L'+str(i)+'"><a class="n" href="#L'+str(i)+'">'+str(i)+'</a><code>'+escape(line)+'</code></span>' for i,line in enumerate(lines,1))
        shutil.copyfile(source, out / (alias + '.source.txt'))
        destination.write_text(page(title, '<a href="index.html">← Audit index</a><h1>'+escape(title)+'</h1><p><code>'+escape(str(source))+'</code></p><small>Source SHA-256: '+sha+'</small><p>Read-only line-numbered text snapshot. <a href="'+alias+'.source.txt">Original text bytes</a>. Relative Markdown links remain text; use the index and professor HTML for navigation and figures.</p><pre>'+content+'</pre>'))
    manifest.append({'alias':alias,'source':str(source),'source_sha256':sha,'view_sha256':hashlib.sha256(destination.read_bytes()).hexdigest()})
    sections.setdefault(group, []).append('<li><a href="'+alias+'.html">'+escape(title)+'</a></li>')

class Links(HTMLParser):
    def handle_starttag(self,tag,attrs):
        if tag=='a':
            href=dict(attrs).get('href')
            assert isinstance(href,str)
            if not href.startswith('#'):
                assert (out/href).is_file(), href

Links().feed((out/'index.html').read_text())
assert len(manifest)==len({row['alias'] for row in manifest})
